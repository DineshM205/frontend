 2. Create `Home`, `About`, and `Users` pages with `<Routes>` and `<Route>`.
//App.jsx
import React from "react";
import { Routes, Route } from "react-router-dom";
import Home from "./pages/Home";
import About from "./pages/About";
import Users from "./pages/Users";
function App() {
  return (
    <div>
      <Routes>
        <Route path="/" element={<Home />} />
        <Route path="/about" element={<About />} />
        <Route path="/users" element={<Users />} />
      </Routes>
    </div>
  );
}
export default App;

 3. Add a `<Link>`-based navbar to navigate between pages.
//components/Navbar.jsx
import React from "react";
import { Link } from "react-router-dom";
function Navbar() {
  return (
    <nav>
      <Link to="/">Home</Link> |{" "}
      <Link to="/about">About</Link> |{" "}
      <Link to="/users">Users</Link>
    </nav>
  );
}
export default Navbar;

 4. Create a dynamic route `/users/:id` and read the id with `useParams`.
//pages/UserDetails.jsx
import React from "react";
import { useParams } from "react-router-dom";

function UserDetails() {
  const { id } = useParams();
  return (
    <div>
      <h2>User Details Page</h2>
      <p>User ID: {id}</p>
    </div>
  );
}
export default UserDetails;
//Update App.jsx
import React from "react";
import { Routes, Route } from "react-router-dom";

import Navbar from "./components/Navbar";

import Home from "./pages/Home";
import About from "./pages/About";
import Users from "./pages/Users";
import UserDetails from "./pages/UserDetails";

function App() {
  return (
    <div>
      <Navbar />
      <Routes>
        <Route path="/" element={<Home />} />
        <Route path="/about" element={<About />} />
        <Route path="/users" element={<Users />} />
        <Route path="/users/:id" element={<UserDetails />} />
      </Routes>
    </div>
  );
}
export default App;

 5. Add a 'Back to Users' button that uses `useNavigate` to go back.
//Update UserDetails.jsx
import React from "react";
import { useParams, useNavigate } from "react-router-dom";

function UserDetails() {
  const { id } = useParams();
  const navigate = useNavigate();
  return (
    <div>
      <h2>User Details Page</h2>
      <p>User ID: {id}</p>
      <button onClick={() => navigate("/users")}>
        Back to Users
      </button>
    </div>
  );
}
export default UserDetails;
